# Ver qué ha cambiado y qué está preparado (úsalo antes de cada add)
git status

# Preparar todos los cambios para el commit (o: git add archivo.py)
git add .

# Guardar la "foto" con un mensaje que explique lo que hiciste
git commit -m "Primer commit: ejercicios de Python y API de tareas"

# Comprobar que no queda nada pendiente ("nothing to commit")
git status

# Ver el historial de commits
git log


# ===== SUBIR A GITHUB =====

# [FALTA] Antes: crear el repositorio vacío en github.com (New repository).
#   NO marcar README, .gitignore ni licencia.

# Conectar tu carpeta con el repositorio de GitHub (UNA sola vez)
git remote add origin https://github.com/JuanEliasLaraSoto/practicando-python.git

# [FALTA] Comprobar que quedó bien guardado
git remote -v

# [FALTA] Si sale "remote origin already exists":
#   corrige la URL con:   git remote set-url origin <url>
#   o bórralo con:        git remote remove origin

# Renombrar la rama principal a "main" (nombre estándar en GitHub)
git branch -M main

# Subir los commits. El -u recuerda el destino: luego basta con "git push"
git push -u origin main

# [FALTA] Comprobar en la web que aparecen tus archivos
#   y que tareas.db NO está.
